cmake_minimum_required(VERSION 3.16)
project(Ink4Preview LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 17)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory(../../build-firmware/managed_components/lvgl__lvgl lvgl)
target_include_directories(lvgl PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../build-firmware/main/display")
set(APP "${CMAKE_CURRENT_SOURCE_DIR}/../../firmware/codex_monitor")
file(GLOB FONTS "${APP}/fonts/codex_inter_*.c")
add_executable(preview main.cpp "${APP}/view/quota_view.cpp" "${APP}/view/codex_icon.c" ${FONTS})
target_link_libraries(preview lvgl m)
